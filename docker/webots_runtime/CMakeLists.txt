cmake_minimum_required(VERSION 3.16)
project(robotics_webots_runtime)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(JAZZY PLUGINLIB__DISABLE_BOOST_FUNCTIONS)
find_package(ament_cmake REQUIRED)
find_package(Python3 3.12 EXACT REQUIRED COMPONENTS Development)
find_package(yaml-cpp REQUIRED)
set(DEPS rclcpp rosgraph_msgs pluginlib sensor_msgs std_msgs tf2_geometry_msgs tf2_ros vision_msgs webots_ros2_msgs tinyxml2_vendor TinyXML2 controller_manager hardware_interface)
foreach(DEP ${DEPS})
 find_package(${DEP} REQUIRED)
endforeach()
add_compile_definitions(HARDWARE_INTERFACE_VERSION_MAJOR=${hardware_interface_VERSION_MAJOR} HARDWARE_INTERFACE_VERSION_MINOR=${hardware_interface_VERSION_MINOR})
file(GLOB STATIC_SOURCES webots_ros2_driver/src/plugins/static/*.cpp)
add_executable(driver webots_ros2_driver/src/Driver.cpp webots_ros2_driver/src/WebotsNode.cpp webots_ros2_driver/src/PythonPlugin.cpp webots_ros2_driver/src/plugins/Ros2SensorPlugin.cpp webots_ros2_driver/src/utils/Math.cpp webots_ros2_driver/src/utils/Utils.cpp webots_ros2_control/src/Ros2Control.cpp ${STATIC_SOURCES})
target_include_directories(driver PRIVATE webots_ros2_driver/include webots_ros2_control/include /opt/ros/jazzy/include/webots/c /opt/ros/jazzy/include/webots/cpp ${Python3_INCLUDE_DIRS})
ament_target_dependencies(driver ${DEPS})
target_link_libraries(driver /opt/ros/jazzy/lib/controller/libController.so /opt/ros/jazzy/lib/controller/libdriver.so /opt/ros/jazzy/lib/controller/libcar.so ${Python3_LIBRARIES} yaml-cpp)
set_target_properties(driver PROPERTIES BUILD_RPATH "/opt/ros/jazzy/lib/controller")
